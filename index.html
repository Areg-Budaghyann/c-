C++ հուշաթերթ. զանգվածներ, մատրիցներ, ֆունկցիաներ

Բոլոր խնդիրները կառուցվում են երեք ձևանմուշից

Յուրաքանչյուր բլոկում կոդը գրեթե նույնն է։ Խնդրից խնդիր փոխվում են միայն երեք տեղ՝ սկզբնական արժեքը, պայմանը և գործողությունը։ Սովորիր կմախքը, իսկ այդ երեք տեղերը լրացրու ստորև բերված աղյուսակներից։

Ամենուր համարակալումը սկսվում է 1-ից, ինչպես խնդրագրքում. զանգվածը հայտարարված է պաշարով՝ x[101], 0-րդ բջիջը չի օգտագործվում։ for ցիկլը բլոկ-սխեմայում պատկերվում է վեցանկյունով՝ i = 1, n, այսինքն՝ «i-ն 1-ից մինչև n»։

օվալ — սկիզբ, վերջ, վերադարձ զուգահեռագիծ — ներմուծում, արտածում ուղղանկյուն — հաշվարկ շեղանկյուն — պայման վեցանկյուն — for ցիկլ կողային գծերով ուղղանկյուն — ֆունկցիայի կանչ

Ձևանմուշ 1. Միաչափ զանգված (221–260, 291–300)

զանգվածի ներմուծում → ցիկլ ըստ i-ի → if (պայման) → գործողություն → արտածում
#include <iostream>
#include <cmath>
using namespace std;

int main() {
    int n;
    double x[101];      // int x[101], եթե պետք են % և զույգություն
    արդյունքի փոփոխականներ
    cin >> n;
    for (int i = 1; i <= n; i++)
        cin >> x[i];

    սկզբնական արժեք;
    for (int i = 1; i <= n; i++)
        if (պայման)
            գործողություն;

    արտածում;
    return 0;
}

Ինչ ենք կուտակում (սկզբնական արժեքը, գործողությունը if-ի ներսում, ինչ ենք արտածում).

Պետք է գտնելՑիկլից առաջՑիկլում (if-ի ներսում)Արտածում
Գումարըs = 0s = s + x[i]s
Արտադրյալըp = 1p = p * x[i]p

Միջին քառակուսայինըs = 0, cnt = 0s = s + x[i] * x[i], cnt = cnt + 1sqrt(s / cnt)
Քառակուսիների, խորանարդների, բացարձակ արժեքների գումարըs = 0s = s + x[i] * x[i], x[i] * x[i] * x[i], fabs(x[i])s
Նոր Y վեկտոր (291–300)m = 0m = m + 1, y[m] = x[i]ցիկլ i = 1, m. y[i]

Ինչ պայման գրել if-ում.

Խնդրում ասված էՊայմանը
ինդեքսը զույգ է / կենտ էi % 2 == 0 / i % 2 != 0
ինդեքսը բազմապատիկ է k-ինi % k == 0

տարրը զույգ է / կենտ է (ամբողջ թվեր)x[i] % 2 == 0 / x[i] % 2 != 0

Ինչու է այդպես

  • Գումարը սկսում ենք 0-ից, իսկ արտադրյալը՝ 1-ից. եթե p = 0, ցանկացած բազմապատկումից հետո կմնա 0։
  • Միջինի համար պետք է երկու փոփոխական՝ գումարը s և քանակը cnt, որովհետև միջին = գումար / քանակ։ Բաժանելուց առաջ ստուգում ենք cnt > 0, այլապես կբաժանենք 0-ի։
  • Միջին քառակուսայինը. գումարում ենք քառակուսիները, բաժանում ենք քանակի վրա և արմատ հանում՝ sqrt(s / cnt)։ sqrt-ի, fabs-ի, sin-ի, log-ի համար պետք է #include <cmath>։
  • s-ը միշտ double է։ Եթե s-ը և cnt-ը երկուսն էլ int լինեն, s / cnt-ը կկտրի կոտորակային մասը (7 / 2 = 3)։
  • % (մնացորդ) գործողությունն աշխատում է միայն ամբողջ թվերի հետ։ Եթե խնդիրը խոսում է զույգության, բազմապատիկության կամ մնացորդի մասին, զանգվածը հայտարարիր int։ Իրական թվի ամբողջ մասը՝ (int)x[i]։
  • Ինդեքսի մասին պայմանը («զույգ ինդեքսով», «ինդեքսը բազմապատիկ է k-ին») գրվում է i-ով, իսկ արժեքի մասին՝ x[i]-ով։ Այս երկուսը շփոթելը ամենահաճախ սխալն է։
  • «Բազմապատիկ է k-ին» և «բաժանվում է k-ի առանց մնացորդի» նույն բանն են՝ % k == 0։
  • Երբ պետք է երկու պատասխան (230, 243, 246), ցիկլը մեկն է, կուտակիչները՝ երկու. կամ մեկ if-ի մեջ երկու գործողություն, կամ երկու առանձին if։

Ձևանմուշ 1բ. Մեծագույն և փոքրագույն (261–270)

Նույն ցիկլն է, բայց սկսում ենք առաջին տարրից և անցնում i = 2-ից։ Ամենահարմարը k համարը պահելն է. այդ դեպքում արժեքը x[k]-ն է։

k = 1;
for (int i = 2; i <= n; i++)
    if (x[i] > x[k])
        k = i;
// x[k]՝ արժեքը, k՝ համարը
Պետք էՆշանը if-ում
մեծագույնը (առաջինը)x[i] > x[k]
վերջին մեծագույնըx[i] >= x[k]
փոքրագույնը (առաջինը)x[i] < x[k]
վերջին փոքրագույնըx[i] <= x[k]

Ինչու է այդպես

  • Սկզբնական արժեքը վերցնում ենք առաջին տարրից, ոչ թե 0-ից. եթե բոլոր տարրերը բացասական են, 0-ն սխալ «մեծագույն» կդառնա։ Քանի որ առաջինն արդեն վերցված է, ցիկլը սկսվում է 2-ից։
  • Եթե պետք է և՛ արժեքը, և՛ համարը, պահիր միայն k համարը, արժեքը x[k]-ն է։
  • > նշանով k-ն փոխվում է միայն խիստ մեծ տարր գտնելիս, ուստի մնում է առաջին մեծագույնը։ >= նշանով այն փոխվում է նաև հավասարի դեպքում, ուստի մնում է վերջինը։
  • Մեծագույնն ու փոքրագույնը միասին (263, 264). մեկ ցիկլ, երկու իրարից անկախ if։

Ձևանմուշ 1գ. Նոր վեկտոր (291–300)

m = 0;
for (int i = 1; i <= n; i++)
    if (պայման) {
        m = m + 1;
        y[m] = x[i];
    }
if (m == 0)
    cout << "Վեկտորը դատարկ է";
else
    for (int i = 1; i <= m; i++)
        cout << y[i] << " ";

Ինչու է այդպես

  • m-ը միաժամանակ նոր վեկտորի տարրերի քանակն է և վերջին լցված բջջի համարը։ Սկզբում մեծացնում ենք m-ը, հետո գրում y[m]-ի մեջ, այդպես առաջին տարրը կընկնի y[1]-ի մեջ։
  • Ցիկլում x-ի համարը i-ն է, իսկ y-ի համարը՝ m-ը. դրանք տարբեր են, որովհետև ոչ բոլոր տարրերն են անցնում։
  • «Հեռացնելով» բառով խնդիրներում պայմանը շրջվում է. «հեռացնել դրականները» = «վերցնել x[i] <= 0»։
  • Եթե ոչ մի տարր չհամապատասխանեց, m-ը մնում է 0, և արտածում ենք «Վեկտորը դատարկ է»։

Ձևանմուշ 2. n×n մատրից (411–440)

մատրիցի ներմուծում → ցիկլ ըստ i-ի → ցիկլ ըստ j-ի → if (տիրույթ && պայման) → գործողություն → արտածում

Սա 1-ին ձևանմուշն է, միայն ցիկլերը երկուսն են, իսկ պայմանը բաղկացած է երկու մասից՝ որտեղ է գտնվում տարրը (տիրույթ) և ինչպիսին է այն (պայման 1-ին ձևանմուշի աղյուսակից, որտեղ x[i]-ն փոխարինվում է c[i][j]-ով)։ Գործողությունները նույնն են՝ գումար, արտադրյալ, քանակ, միջիններ։

int n;
double c[11][11];
cin >> n;
for (int i = 1; i <= n; i++)
    for (int j = 1; j <= n; j++)
        cin >> c[i][j];

սկզբնական արժեք;
for (int i = 1; i <= n; i++)
    for (int j = 1; j <= n; j++)
        if (տիրույթ && պայման)
            գործողություն;
արտածում;

Տիրույթներ. Սեղմիր՝ բջիջները տեսնելու համար։ Գլխավոր անկյունագիծը գնում է վերին ձախ անկյունից դեպի ստորին աջը, օժանդակը՝ վերին աջից դեպի ստորին ձախը։

n×n մատրիցում, այստեղ n = 6

ՏիրույթԳլխավոր անկյունագիծՕժանդակ անկյունագիծ
անկյունագծի վրաi == ji + j == n + 1
վերևi < ji + j < n + 1
վերև կամ նրա վրաi <= ji + j <= n + 1
ներքևi > ji + j > n + 1
ներքև կամ նրա վրաi >= ji + j >= n + 1

Եվս երեք պայման, որոնք հանդիպում են միայն մատրիցներում. ինդեքսների գումարը կենտ է՝ (i + j) % 2 != 0, տարրն ամբողջ թիվ է՝ c[i][j] == (int)c[i][j], կոտորակային մասը՝ c[i][j] - (int)c[i][j]։ 411–420 խնդիրներում չափը նշանակված է m, դա նույնն է, ինչ n-ը։

Ինչու է այդպես

  • i-ն տողի համարն է (վերևից ներքև), j-ն՝ սյան համարը (ձախից աջ)։
  • Գլխավոր անկյունագծի վրա տողի և սյան համարները հավասար են՝ i == j։ Նրանից վերև սյան համարն ավելի մեծ է՝ i < j, ներքև՝ ավելի փոքր՝ i > j։
  • Օժանդակ անկյունագծի վրա i + j-ն միշտ հավասար է n + 1-ի. n = 4-ի դեպքում դրանք (1,4), (2,3), (3,2), (4,1) բջիջներն են։ Վերևում գումարն ավելի փոքր է, ներքևում՝ ավելի մեծ։
  • «Վերև» կամ «ներքև»՝ խիստ նշան (<, >)։ «Վերև կամ նրա վրա»՝ ոչ խիստ (<=, >=)։
  • Տիրույթն ու պայմանը միացնում ենք &&-ով («և»)։ Ամբողջ մատրիցն անցնում ենք երկու լրիվ ցիկլով, իսկ ավելորդ բջիջները կտրում է if-ը. այդպես հնարավոր չէ սխալվել ցիկլի սահմաններում։
  • Մեծագույն կամ փոքրագույն գտնելիս սկզբնական արժեքը վերցրու տիրույթի ներսում գտնվող բջջից. օրինակ՝ գլխավոր անկյունագծի համար c[1][1], օժանդակի համար՝ c[1][n]։
  • Բլոկ-սխեմայում j-ի ցիկլն ամբողջությամբ գտնվում է i-ի ցիկլի ներսում, իսկ if-ը՝ j-ի ցիկլի ներսում։

Ձևանմուշ 3. Ֆունկցիաներ և պրոցեդուրաներ (591–625)

ֆունկցիան պատասխանը տալիս է return-ով, պրոցեդուրան void է և պատասխանը տալիս է & նշանով պարամետրի միջոցով
ՖունկցիաՊրոցեդուրա
Վերնագիրdouble max3(double x, double y, double z)void max3(double x, double y, double z, double &m)
Ինչպես է տալիս պատասխանըreturn m;գրում է m-ի մեջ (&-ի շնորհիվ փոխվում է main-ի փոփոխականը)
Կանչը main-ումm1 = max3(a, a + b, a - b);max3(a, a + b, a - b, m1);
Վերջը սխեմայումօվալ «Վերադարձ m»օվալ «Վերջ»

591–600. Մեկ անգամ գրում ես max3 կամ min3 (min-ի համար նշանը < է), իսկ main-ում յուրաքանչյուր փակագիծ հաշվում ես առանձին կանչով՝ m1 = …, m2 = …, հետո՝ y-ը։

double max3(double x, double y, double z) {
    double m;
    m = x;
    if (y > m) m = y;
    if (z > m) m = z;
    return m;
}

void max3p(double x, double y, double z, double &m) {
    m = x;
    if (y > m) m = y;
    if (z > m) m = z;
}
max3 ֆունկցիա
Հիմնական ծրագիր (591)

601–610. φ ֆունկցիան մեկ տող է՝ return բանաձև;։ main-ում x-ը և y-ը տալիս ես պայմանից, ներմուծում ես a-ն, հաշվում ես f = phi(x, y), հետո՝ z-ը։ 606–610-ում նույնը պրոցեդուրայով է՝ void phi(double x, double y, double &f) { f = բանաձև; }։ Լոգարիթմը 3 հիմքով՝ log(t) / log(3), ctg՝ 1 / tan(t), arctg՝ atan(t)։

double phi(double x, double y) {
    return log2(x + y);
}
// main-ում.
x = 2; y = 5;
cin >> a;
f = phi(x, y);
z = a * (x * x + y * y) * f * f;

611–625. Զանգվածի ներմուծումը միշտ input պրոցեդուրան է, արտածումը՝ output պրոցեդուրան, իսկ «թիվ հաշվելը»՝ ֆունկցիա, որի մարմինը 1-ին ձևանմուշն է։ Զանգվածը փոխանցվում է առանց &-ի (այն առանց այդ էլ փոխվում է), իսկ n-ը՝ &-ով, որովհետև ներմուծվում է ներսում։

void input(int &n, double x[]) {
    cin >> n;
    for (int i = 1; i <= n; i++)
        cin >> x[i];
}

void output(int m, double y[]) {
    for (int i = 1; i <= m; i++)
        cout << y[i] << " ";
}

double sumPos(int n, double x[]) {
    double s;
    s = 0;
    for (int i = 1; i <= n; i++)
        if (x[i] > 0)
            s = s + x[i];
    return s;
}
// main-ում.
input(n, x);
s = sumPos(n, x);
input պրոցեդուրա
Հիմնական ծրագիր

Ինչու է այդպես

  • Ֆունկցիա ենք գրում, երբ արդյունքը մեկ թիվ է, որը միանգամից օգտագործվում է արտահայտության մեջ՝ y = m1 + m2։ Վերնագրում տիպը (double կամ int) այն է, ինչ վերադարձնում է return-ը։
  • Պրոցեդուրան (void) ոչինչ չի վերադարձնում։ Պատասխանը գրվում է այն պարամետրի մեջ, որի առջև կա &։ Առանց &-ի պրոցեդուրան կփոխի իր պատճենը, և main-ում արժեքը չի փոխվի։
  • Զանգվածը (x[]) և մատրիցը (c[][11]) փոխանցվում են առանց &-ի, և ենթածրագրում կատարված փոփոխությունները միևնույն է երևում են main-ում։
  • Ենթածրագիրը գրվում է main-ից վերև, որպեսզի կոմպիլյատորն այն ճանաչի կանչելու պահին։
  • Պարամետրերի անունները կարող են չհամընկնել main-ի փոփոխականների անունների հետ. կարևոր է միայն հերթականությունը։ max3(a, a + b, a - b) կանչում x = a, y = a + b, z = a - b։
  • Բլոկ-սխեմայում յուրաքանչյուր ենթածրագիր ունի իր առանձին սխեման։ Ֆունկցիայինն ավարտվում է «Վերադարձ …»-ով, պրոցեդուրայինը՝ «Վերջ»-ով։ main-ում կանչը գծվում է կողային գծերով ուղղանկյունով։

Տիպիկ սխալներ

  • Արտադրյալի համար p = 0 (պետք է p = 1)։
  • Ներմուծումը 1-ից, իսկ մշակման ցիկլը 0-ից (կամ հակառակը)։ Ամենուր օգտագործիր 1 … n։
  • % double թվի հետ՝ կոմպիլյացիայի սխալ։ Օգտագործիր int զանգված կամ (int)x[i]։
  • = (վերագրում) և == (համեմատում) շփոթել if-ի մեջ։
  • Միջինը առանց cnt > 0 ստուգման։
  • Մոռացված & պրոցեդուրայի ելքային պարամետրի առջև։
  • Մոռացված #include <cmath>, երբ կոդում կա sqrt, fabs, sin, log։
  • Մեծագույնի սկզբնական արժեքը 0, ոչ թե առաջին տարրը։
